kernel=Image

# Disable overscan assuming the display supports displaying the full resolution
disable_overscan=1

# WM1302 Pi HAT: the SX1302 is on SPI0 CE0 (/dev/spidev0.0) and its temperature sensor is on
# I2C1, header pins 3 and 5 (/dev/i2c-1). i2c-dev is loaded at boot by /etc/modules-load.d/wm1302.conf.
dtparam=spi=on
dtparam=i2c_arm=on
